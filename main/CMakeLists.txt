file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")

# Game content lives in the project-root assets/ folder (CaptureIt/assets), next to
# main/ rather than inside it, so the Game Tools asset explorer can hold the map
# editor's own art there too. Only the parts read below are built into the firmware.
get_filename_component(GAME_ASSETS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../assets" ABSOLUTE)

# --- Items: every assets/items/<id>.c is one item -------------------
# Written by the Game Tools Item Editor. The file name IS the item id: it
# must define `const item_asset_t k_item_<id>` (the editor does this). This
# plain glob only feeds SRCS — IDF's requirements pre-pass runs this file
# in script mode and stops at idf_component_register(), so the
# CONFIGURE_DEPENDS glob + manifest generation live below it.
file(GLOB ITEM_SRCS "${GAME_ASSETS_DIR}/items/*.c")
# Crops: every assets/crops/<id>.c (Game Tools Crops Editor) is one crop
# PLUS its two items, k_item_<id> (the harvest) and k_item_<id>_seeds.
file(GLOB CROP_SRCS "${GAME_ASSETS_DIR}/crops/*.c")
# Buildings: every assets/buildings/<id>.c (Game Tools Buildings Editor) is one
# building PLUS the item that places it, k_item_<id>.
file(GLOB BUILDING_SRCS "${GAME_ASSETS_DIR}/buildings/*.c")
# Animals: every assets/animals/<id>.c (Game Tools Animals Editor) is one farm
# animal PLUS the item that lets it out, k_item_<id>.
file(GLOB ANIMAL_SRCS "${GAME_ASSETS_DIR}/animals/*.c")
# Machines: every assets/machines/<id>.c (Game Tools Machines Editor) is one machine
# (keg, cheese press...) PLUS the prop item that places it, k_item_<id>.
file(GLOB MACHINE_SRCS "${GAME_ASSETS_DIR}/machines/*.c")
# Fences: every assets/fences/<id>.c (Game Tools Farm tab, Fences & pipes) is one fence,
# gate or pipe PLUS the prop item that places it, k_item_<id>.
file(GLOB FENCE_SRCS "${GAME_ASSETS_DIR}/fences/*.c")
# Fishing: assets/fishing/settings.c (Game Tools' Fishing tab) holds k_fishing_cfg, the
# game-wide fishing settings (items.h fishing_cfg_t). It must be there: the link needs it.
file(GLOB FISHING_SRCS "${GAME_ASSETS_DIR}/fishing/*.c")
# Trees: every assets/trees/<id>.c (Game Tools Trees Editor) is one tree PLUS
# the item it's planted from, k_item_<id>_seed.
file(GLOB TREE_SRCS "${GAME_ASSETS_DIR}/trees/*.c")

idf_component_register(SRCS "items.c" "crops.c" "buildings.c" "animals.c" "machines.c" "fences.c" "trees.c" "actions.c" "main.c" ${ITEM_SRCS} ${CROP_SRCS} ${BUILDING_SRCS} ${ANIMAL_SRCS} ${MACHINE_SRCS} ${FENCE_SRCS} ${FISHING_SRCS} ${TREE_SRCS}
                        INCLUDE_DIRS "." "${CMAKE_CURRENT_BINARY_DIR}/generated")

# Re-glob with CONFIGURE_DEPENDS so adding/removing an item file re-runs
# CMake on the next build (no manual reconfigure), then write
# items_manifest.inc — one ITEM_ENTRY(<id>) per file, sorted by name.
# items.c expands it into the s_items[] table. Only rewritten when the list
# changes, so unrelated builds don't recompile items.c.
file(GLOB ITEM_SRCS_WATCH CONFIGURE_DEPENDS "${GAME_ASSETS_DIR}/items/*.c")
list(SORT ITEM_SRCS_WATCH)
set(ITEM_MANIFEST "// AUTO-GENERATED by main/CMakeLists.txt from assets/items/*.c — do not edit.\n")
set(ITEM_IDS_SEEN "")
foreach(item_src ${ITEM_SRCS_WATCH})
    get_filename_component(item_id "${item_src}" NAME_WE)
    if(NOT item_id MATCHES "^[a-z_][a-z0-9_]*$")
        message(FATAL_ERROR "Item file '${item_src}': the file name is the item id and must be "
                            "lowercase letters, digits and underscores (re-export it from the Item Editor).")
    endif()
    if(item_id STREQUAL "none")
        message(FATAL_ERROR "Item file '${item_src}': 'none' is reserved for the empty slot.")
    endif()
    list(APPEND ITEM_IDS_SEEN "${item_id}")
    string(APPEND ITEM_MANIFEST "ITEM_ENTRY(${item_id})\n")
endforeach()

# Crops: crops_manifest.inc gets CROP_ENTRY(<id>) per file, and the item
# manifest gets the crop's own two items. An id clash with an items/ file
# (or another crop's seeds) stops the build here instead of at link time.
file(GLOB CROP_SRCS_WATCH CONFIGURE_DEPENDS "${GAME_ASSETS_DIR}/crops/*.c")
list(SORT CROP_SRCS_WATCH)
set(CROP_MANIFEST "// AUTO-GENERATED by main/CMakeLists.txt from assets/crops/*.c — do not edit.\n")
set(CROP_IDS_SEEN "")
foreach(crop_src ${CROP_SRCS_WATCH})
    get_filename_component(crop_id "${crop_src}" NAME_WE)
    if(NOT crop_id MATCHES "^[a-z_][a-z0-9_]*$")
        message(FATAL_ERROR "Crop file '${crop_src}': the file name is the crop id and must be "
                            "lowercase letters, digits and underscores (re-export it from the Crops Editor).")
    endif()
    foreach(crop_item "${crop_id}" "${crop_id}_seeds")
        list(FIND ITEM_IDS_SEEN "${crop_item}" crop_clash)
        if(NOT crop_clash EQUAL -1 OR crop_item STREQUAL "none")
            message(FATAL_ERROR "Crop file '${crop_src}' makes the item '${crop_item}', but that id is "
                                "already taken (assets/items/${crop_item}.c or another crop). Rename one of them.")
        endif()
        list(APPEND ITEM_IDS_SEEN "${crop_item}")
        string(APPEND ITEM_MANIFEST "ITEM_ENTRY(${crop_item})\n")
    endforeach()
    list(APPEND CROP_IDS_SEEN "${crop_id}")
    string(APPEND CROP_MANIFEST "CROP_ENTRY(${crop_id})\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/crops_manifest.inc"
     CONTENT "${CROP_MANIFEST}" @ONLY)

# --- PNG -> RGB565 sprite generation --------------------------------------
# Regenerates main/generated/*.h from every PNG in assets/ before
# main.c compiles. Needs Pillow in the same Python env idf.py uses:
#   pip install pillow --break-system-packages
find_package(Python3 COMPONENTS Interpreter REQUIRED)

set(SPRITE_SRC_DIR "${GAME_ASSETS_DIR}")
set(SPRITE_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(SPRITE_SCRIPT  "${CMAKE_CURRENT_SOURCE_DIR}/../tools/png_to_rgb565.py")
# Top-level sprites, ui/ (title screen art), fx/, and wardrobe/ (everyone's
# looks, in layers: tools/wardrobe.py). CONFIGURE_DEPENDS: a new PNG
# re-runs CMake, so it's picked up without a manual reconfigure.
file(GLOB SPRITE_PNGS CONFIGURE_DEPENDS "${SPRITE_SRC_DIR}/*.png" "${SPRITE_SRC_DIR}/ui/*.png" "${SPRITE_SRC_DIR}/fx/*.png")
file(GLOB_RECURSE WARDROBE_PNGS CONFIGURE_DEPENDS "${SPRITE_SRC_DIR}/wardrobe/*.png")

add_custom_command(
    OUTPUT  "${SPRITE_GEN_DIR}/sprites_gen.h"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${SPRITE_GEN_DIR}"
    COMMAND ${Python3_EXECUTABLE} "${SPRITE_SCRIPT}"
            --in-dir "${SPRITE_SRC_DIR}" --out-dir "${SPRITE_GEN_DIR}"
    DEPENDS ${SPRITE_PNGS} ${WARDROBE_PNGS} "${SPRITE_SCRIPT}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/wardrobe.py"
    COMMENT "Converting sprite PNGs to RGB565 headers"
)
add_custom_target(sprites_gen DEPENDS "${SPRITE_GEN_DIR}/sprites_gen.h")
add_dependencies(${COMPONENT_LIB} sprites_gen)
